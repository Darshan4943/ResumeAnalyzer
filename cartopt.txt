import React, { useState, useEffect, useMemo, useReducer } from "react";
import { connect, useDispatch, useSelector } from "react-redux";
import ALink from "~/components/features/alink";
import PageHeader from "~/components/features/page-header";
import Select from "react-select";
import { toast } from "react-toastify";
import DatePicker from "react-date-picker/dist/entry.nostyle";
import "react-date-picker/dist/DatePicker.css";
import "react-calendar/dist/Calendar.css";
import { brandSlider, fadeIn } from "~/utils/data";
import Reveal from "react-awesome-reveal";
import OwlCarousel from "~/components/features/owl-carousel";
import axios from "axios";
import { useRouter } from "next/router";
import { incNumber } from "~/Redux/actions";
import { addToCart, localAddToCart } from "~/utils/services";
// import DatePicker from "react-datepicker";
// import "react-datepicker/dist/react-datepicker.css";

function Cart(props) {
  const [shippingCost, setShippingCost] = useState(0);
  const [requiredData, setRequiredData] = useState(new Date());
  const router = useRouter();
  const recall = useSelector((state) => state.changeNumber);
  const carts = [];
  const dispatch = useDispatch();
  const userData = JSON.parse(localStorage.getItem("userResponse"));
  const allProducts = useSelector((state) => state.allProducts.products);
  const cartData = useSelector((state) => state.allCart.carts);
  const cartType = JSON.parse(localStorage.getItem("cartType"));
  const [reduceValue, forceUpdate] = useReducer((x) => x + 1, 0);
  if (!cartData.length == 0) {
    cartData.forEach((item) => {
      if (item.type === cartType?.cartType) {
        const product = allProducts?.find(
          (product) => product._id === item.productId
        );
        carts.push({
          ...product,
          quantity: item.productQuantity,
          cartId: item._id,
          cartIdentifier: item.cartIdentifier,
        });
      }
    });
  }
  let subtotal = 0;
  carts.forEach((item) => {
    subtotal += item.salePrice * item.quantity;
  });

  const updateCart = (id, obj) => {
    axios
      .put(`https://freedygoservices.in/api/cart/update/${id}`, obj)
      .then((res) => {
        console.log(res);
        dispatch(incNumber());
      });
  };

  const removeFromCart = (e, id) => {
    e.preventDefault();
    axios
      .delete(`https://freedygoservices.in/api/cart/delete/${id}`)
      .then((res) => {
        console.log(res);
        dispatch(incNumber());
        toast.success("Product deleted successfully");
      })
      .catch((err) => console.log(err));
  };
  function increment(e, current, id) {
    e.preventDefault();
    const obj = {
      productQuantity: current + 1,
    };
    updateCart(id, obj);
  }

  function decrement(e, current, id) {
    e.preventDefault();
    if (current > 1) {
      const obj = {
        productQuantity: current - 1,
      };
      updateCart(id, obj);
    }
  }

  function onChangeShipping(value) {
    setShippingCost(value);
  }

  const onChange = (value) => {
    const selecteDate = new Date(value);
    const selecteDay = new Date(value).getDate();

    console.log(new Date(value).getDate());
    const TodayDate = new Date().getDate();
    if (selecteDay < TodayDate) {
      toast.error("Please Select Correct Date.....");
    } else if (selecteDate) {
      setRequiredData((old) => ({ ...old, date: selecteDate }));
    } else {
      setRequiredData((old) => ({ ...old, date: new Date() }));
    }
  };
  const PlaceOrder = () => {
    //console.log("hello 205")
    localStorage.setItem(
      "orderData",
      JSON.stringify({
        date: requiredData.date,
        // location: requiredData.pincode.__typename,
        pincode: requiredData.pincode.lable,

        time: requiredData.time.value,
        subtotal,
        shippingCost,
      })
    );
    router.push("/shop/checkout");
  };
  const [user, setUser] = useState(null);
  const [productToOptimize, setproductToOptimize] = useState([]);
  useEffect(() => {
    const data = JSON.parse(localStorage.getItem("productToOptimized"));
    if (data) {
      setproductToOptimize(data);
    }
    const userData = JSON.parse(localStorage.getItem("userResponse"));
    if (userData) {
      setUser(userData);
    }
  }, [reduceValue]);
  console.log("user", user);
  const isCheck = (arr, pId) => {
    const find = productToOptimize.find((item) => item.cId === pId);
    if (find) {
      const firstValue = Object.values(find)[0];
      if (firstValue.length > 0) {
        return true;
      } else {
        false;
      }
    }
  };
  const isDisabled = (fc, pId) => {
    const filter = allProducts
      .filter((item) => item.familyCode === fc)
      .filter((data) => {
        if (data._id != pId) {
          return data;
        }
      });
    if (filter.length > 0) {
      return false;
    } else {
      return true;
    }
  };
  const upValue = (data) => {
    const maxAmount = 0;
    const minAmount = 0;
    data.forEach((item) => {
      let values = Object.values(item);
      let key = Object.keys(item);
      // const currentCart = carts.find(item=>item.familyCode === key[0])
      const amounts = values[0].map((amount) => amount.salePrice * values[1]);
      const max = Math.max(...amounts);
      const min = Math.min(...amounts);
      maxAmount = maxAmount + max;
      minAmount = minAmount + min;
    });
    return { maxAmount, minAmount };
  };
  const changeHandler = (fc, pId, qty, id) => {
    const find = productToOptimize.find((item) => item.cId === pId);
    if (find) {
      const index = productToOptimize.findIndex((object) => {
        return object.cId === pId;
      });
      productToOptimize.splice(index, 1);
      // console.log(201,productToOptimize)
      // forceUpdate()
      setproductToOptimize(productToOptimize);
      localStorage.setItem(
        "productToOptimized",
        JSON.stringify(productToOptimize)
      );
      forceUpdate();
    } else {
      const filter = allProducts
        .filter((item) => item.familyCode === fc)
        .filter((data) => {
          if (data._id != id) {
            return data._id;
          }
        })
        .sort((a, b) => {
          return a.salePrice - b.salePrice;
        });
      let key = fc;
      var obj = {};
      obj[key] = filter;
      obj["quantity"] = qty;
      obj["cId"] = pId;
      if (filter.length > 0) {
        localStorage.setItem(
          "productToOptimized",
          JSON.stringify([...productToOptimize, obj])
        );
        forceUpdate();
      }
    }
  };
  console.log(187, productToOptimize);

  const downOptimize = () => {
    productToOptimize.forEach((item) => {
      let cid = item.cId;
      let optData = Object.values(item)[0];
      const foundCart = carts.find((c) => c.cartId === cid);
      const findMinPrice = optData
        .filter((data) => data.salePrice <= foundCart.salePrice)
        .sort((a, b) => {
          return a.salePrice - b.salePrice;
        });
      const productToUpdate = findMinPrice[findMinPrice.length - 1];
      if (productToUpdate != undefined) {
        const obj = {
          productId: productToUpdate._id,
          productQuantity: productToUpdate.quantity,
          cartIdentifier: productToUpdate._id + user.user._id,
        };
        axios
          .put(
            `https://freedygoservices.in/api/cart/update/${foundCart.cartIdentifier}`,
            obj
          )
          .then((res) => {
            console.log(237, res.data);
          })
          .catch((err) => console.log(err));
      }
    });
    dispatch(incNumber());
    forceUpdate()
  };
  const upOptimize = () => {
    productToOptimize.forEach((item) => {
      let cid = item.cId;
      let optData = Object.values(item)[0];
      const foundCart = carts.find((c) => c.cartId === cid);
      const findMinPrice = optData
        .filter((data) => data.salePrice >= foundCart.salePrice)
        .sort((a, b) => {
          return a.salePrice - b.salePrice;
        });
      const productToUpdate = findMinPrice[0];
      if (productToUpdate != undefined) {
        const obj = {
          productId: productToUpdate._id,
          productQuantity: productToUpdate.quantity,
          cartIdentifier: productToUpdate._id + user.user._id,
        };
        axios
          .put(
            `https://freedygoservices.in/api/cart/update/${foundCart.cartIdentifier}`,
            obj
          )
          .then((res) => {
            console.log(237, res.data);
          })
          .catch((err) => console.log(err));
      }
    });
    dispatch(incNumber());
    forceUpdate()

  };
  return (
    <div className="Container">
      <PageHeader
        title="Shopping Cart"
        subTitle={`Selected Cart : ${cartType.cartType}`}
      />
      <nav className="breadcrumb-nav">
        <div className="container">
          <ol className="breadcrumb">
            <li className="breadcrumb-item">
              <ALink href="/">Home</ALink>
            </li>
            <li className="breadcrumb-item">
              <ALink href="/shop/sidebar/list">Shop</ALink>
            </li>
            <li className="breadcrumb-item active">Shopping Cart</li>
          </ol>
        </div>
      </nav>

      <div className="page-content pb-5">
        <div className="container">
          {carts.length > 0 ? (
            <div className="row">
              <div className="col-lg-8 ">
                <table className="table table-cart table-mobile">
                  <thead>
                    <tr>
                      <th>Product</th>
                      <th className="text-center">Price</th>
                      <th className="text-center">Quantity</th>
                      <th className="text-center">Total</th>
                      <th></th>
                    </tr>
                  </thead>

                  <tbody>
                    {carts.length > 0 ? (
                      carts?.map((item, index) => {
                        return (
                          <tr key={index} style={{ position: "relative" }}>
                            <td className="product-col">
                              <div className="product">
                                <figure className="product-media">
                                  <ALink
                                    href={`/product/default/${item.slug}`}
                                    className="product-image"
                                  >
                                    {item.productImages
                                      ?.slice(0, 1)
                                      .map((img) => {
                                        return (
                                          <img src={img.img} alt="product" />
                                        );
                                      })}
                                  </ALink>
                                </figure>
                                <div className="d-flex">
                                  <h6
                                    className="product-title"
                                    style={{ fontSize: "1.5rem" }}
                                  >
                                    <ALink
                                      href={`/product/default/${item.slug}`}
                                    >
                                      {item.name} , 5kg
                                    </ALink>
                                  </h6>
                                  <h6
                                    className="product-title"
                                    style={{ fontSize: "1.5rem" }}
                                  >
                                    5kg
                                  </h6>
                                </div>
                              </div>
                            </td>
                            <td className="price-col text-center">
                              ₹
                              {item?.salePrice?.toLocaleString(undefined, {
                                minimumFractionDigits: 2,
                                maximumFractionDigits: 2,
                              })}
                            </td>

                            <td className="quantity-col ">
                              <form
                                action=""
                                className="d-flex justify-content-center m-3"
                              >
                                <div className={`product-details-quantity `}>
                                  <div
                                    className="input-group input-spinner"
                                    style={{ width: "113px" }}
                                  >
                                    <div className="input-group-prepend">
                                      <button
                                        style={{ minWidth: "26px" }}
                                        className="btn btn-decrement btn-spinner"
                                        onClick={(e) =>
                                          decrement(
                                            e,
                                            item.quantity,
                                            item.cartIdentifier
                                          )
                                        }
                                        type="submit"
                                      >
                                        <i className="icon-minus"></i>
                                      </button>
                                    </div>
                                    <input
                                      type="number"
                                      className="form-control text-center"
                                      min="1"
                                      value={item.quantity}
                                      required
                                    />
                                    <div
                                      className="input-group-append"
                                      style={{
                                        width: "unset",
                                        background: "none",
                                        padding: "none",
                                      }}
                                    >
                                      <button
                                        style={{ minWidth: "26px" }}
                                        className="btn btn-increment btn-spinner"
                                        type="submit"
                                        onClick={(e) =>
                                          increment(
                                            e,
                                            item.quantity,
                                            item.cartIdentifier
                                          )
                                        }
                                      >
                                        <i className="icon-plus"></i>
                                      </button>
                                    </div>
                                  </div>
                                </div>
                              </form>
                            </td>

                            <td className="total-col ">
                              ₹
                              {(item.salePrice * item.quantity).toLocaleString(
                                undefined,
                                {
                                  minimumFractionDigits: 2,
                                  maximumFractionDigits: 2,
                                }
                              )}
                            </td>

                            <td className="remove-col">
                              <button
                                className="btn-remove"
                                onClick={(e) => removeFromCart(e, item.cartId)}
                              >
                                <i className="icon-close"></i>
                              </button>
                            </td>
                            {isDisabled(item.familyCode, item._id) ? (
                              <td
                                className="optoption"
                                style={{
                                  borderBottom: "unset",
                                  padding: "0px",
                                  width: "203px",
                                  color: "#FF033E",
                                }}
                              >
                                Cost Optimization is Not Available
                              </td>
                            ) : (
                              <td
                                className="optoption"
                                style={{
                                  borderBottom: "unset",
                                  padding: "0px",
                                }}
                              >
                                <input
                                  checked={isCheck(
                                    productToOptimize,
                                    item.cartId
                                  )}
                                  type="checkbox"
                                  style={{
                                    marginRight: "5px",
                                  }}
                                  onChange={() => {
                                    changeHandler(
                                      item.familyCode,
                                      item.cartId,
                                      item.quantity,
                                      item._id
                                    );
                                  }}
                                />
                                Cost Optimization
                              </td>
                            )}
                          </tr>
                        );
                      })
                    ) : (
                      <tr>
                        <td>
                          <p className="pl-2 pt-1 pb-1">
                            {" "}
                            No Products in Cart{" "}
                          </p>
                        </td>
                      </tr>
                    )}
                  </tbody>
                </table>

                <div className="cart-bottom">
                  <div className="cart-discount">
                    <form action="#">
                      <div className="input-group">
                        <input
                          type="text"
                          className="form-control"
                          style={{ margin: "0px" }}
                          required
                          placeholder="coupon code"
                        />
                        <div
                          className="input-group-append"
                          style={{ borderRadius: "unset", padding: "0px" }}
                        >
                          <button
                            className="btn btn-outline-primary-2"
                            type="submit"
                          >
                            <i
                              className="icon-long-arrow-right"
                              style={{ color: "white" }}
                            ></i>
                          </button>
                        </div>
                      </div>
                    </form>
                  </div>
                </div>
              </div>

              <aside className="col-lg-4">
                <div className="summary summary-cart">
                  <h3 className="summary-title">Cart Total</h3>

                  <table className="table table-summary">
                    <tbody>
                      <tr className="summary-subtotal">
                        <td>Subtotal:</td>
                        <td>
                          ₹
                          {subtotal.toLocaleString(undefined, {
                            minimumFractionDigits: 2,
                            maximumFractionDigits: 2,
                          })}
                        </td>
                      </tr>
                      <tr className="summary-subtotal">
                        <td>Discount:</td>
                        <td>
                          ₹
                          {(100).toLocaleString(undefined, {
                            minimumFractionDigits: 2,
                            maximumFractionDigits: 2,
                          })}
                        </td>
                      </tr>
                      <tr className="summary-subtotal">
                        <td>GST:</td>
                        <td>
                          ₹
                          {(100).toLocaleString(undefined, {
                            minimumFractionDigits: 2,
                            maximumFractionDigits: 2,
                          })}
                        </td>
                      </tr>
                      <tr className="summary-total">
                        <td>Total:</td>
                        <td>
                          ₹
                          {(subtotal + shippingCost).toLocaleString(undefined, {
                            minimumFractionDigits: 2,
                            maximumFractionDigits: 2,
                          })}
                        </td>
                      </tr>{" "}
                      <h3
                        className="summary-title"
                        style={{
                          border: "unset",
                          paddingTop: "1.7rem",
                          paddingBottom: "0px",
                        }}
                      >
                        Cost Optimization
                      </h3>
                      <tr className="summary-total">
                        <td>
                          <button
                            className="cartopt mt-2 mb-2"
                            onClick={() => downOptimize()}
                          >
                            Down
                          </button>
                        </td>
                        <td>
                          <button className="cartopt mt-2 mb-2" onClick={()=>upOptimize()}>Up</button>
                        </td>
                      </tr>
                      <tr className="summary-total">
                        <td>
                          <span
                            className="mt-2 mb-2"
                            style={{ color: "black", fontSize: "12px" }}
                          >
                            Optimize amount can go down to{" "}
                            {upValue(productToOptimize).minAmount}
                          </span>
                        </td>
                        <td>
                          <button
                            className="cartopt mt-1 "
                            style={{ width: "35px" }}
                          >
                            {">"}
                          </button>
                        </td>
                      </tr>
                      <tr className="summary-total">
                        <td>
                          <span
                            className=" mb-2"
                            style={{ color: "black", fontSize: "12px" }}
                          >
                            Optimize amount can go up to{" "}
                            {upValue(productToOptimize).maxAmount}
                          </span>
                        </td>
                        <td>
                          <button
                            className="cartopt mt-2 mb-2"
                            style={{ width: "35px" }}
                          >
                            {">"}
                          </button>
                        </td>
                      </tr>
                    </tbody>
                  </table>

                  <a
                    className="btn btn-outline-primary-2 btn-order btn-block"
                    onClick={(e) => {
                      e.preventDefault();
                      router.push("/shop/checkout");
                    }}
                  >
                    PROCEED TO CHECKOUT
                  </a>
                </div>

                <ALink
                  href="#"
                  className="btn btn-outline-dark-2 btn-block mb-3"
                >
                  <span
                    onClick={(e) => {
                      e.preventDefault();
                      router.push({
                        pathname: "/shop/sidebar/list",
                      });
                    }}
                  >
                    {" "}
                    CONTINUE SHOPPING
                  </span>
                  <i className="icon-refresh"></i>
                </ALink>
              </aside>
            </div>
          ) : (
            <div className="row">
              <div className="col-12">
                <div className="cart-empty-page text-center">
                  <i
                    className="cart-empty icon-shopping-cart"
                    style={{ lineHeight: 1, fontSize: "15rem" }}
                  ></i>
                  <p className="px-3 py-2 cart-empty mb-3">
                    No products added to the cart
                  </p>
                  <p
                    className="return-to-shop mb-0"
                    onClick={(e) => {
                      e.preventDefault();
                      router.push({
                        pathname: "/shop/sidebar/list",
                        query: { isAddon: false },
                      });
                    }}
                  >
                    <a>RETURN TO SHOP</a>
                  </p>
                </div>
              </div>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
